cmake_minimum_required(VERSION 3.22)
project(Trait64 VERSION 0.3.0 LANGUAGES CXX)
set(T64_RELEASE_VERSION "${PROJECT_VERSION}-dev.1" CACHE STRING "Published semantic version")
add_compile_definitions(T64_VERSION="${T64_RELEASE_VERSION}")
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(T64_BUILD_GUI "Build the Qt desktop application" ON)
option(T64_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
option(T64_BENCHMARKS "Build opt-in database and real-PGN benchmarks" OFF)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-Wall -Wextra -Wpedantic -Wformat=2)
endif()
if(T64_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
find_package(Threads REQUIRED)
add_library(t64core libs/core/game.cpp libs/core/formats.cpp libs/core/parallel.cpp libs/core/explanation.cpp libs/core/tactics.cpp)
target_link_libraries(t64core PUBLIC Threads::Threads)
target_include_directories(t64core PUBLIC libs/core)
target_include_directories(t64core SYSTEM PUBLIC third_party/chess-library)
include(CTest)
if(BUILD_TESTING)
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE t64core)
  add_test(NAME core COMMAND core_tests)
  add_executable(robustness_tests tests/robustness_tests.cpp)
  target_link_libraries(robustness_tests PRIVATE t64core)
  add_test(NAME robustness COMMAND robustness_tests)
  if(T64_SANITIZERS)
    # The full mutation corpus is CPU-heavy in unoptimized ASan/UBSan builds.
    set_tests_properties(robustness PROPERTIES TIMEOUT 300)
  else()
    set_tests_properties(robustness PROPERTIES TIMEOUT 120)
  endif()
endif()
if(T64_BUILD_GUI)
  find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 Sql Concurrent Test Network)
  find_package(LibArchive REQUIRED)
  find_package(SQLite3 REQUIRED)
  set(CMAKE_AUTOMOC ON)
  add_library(t64storage libs/database/analyses.cpp libs/database/repertoires.cpp libs/database/imports.cpp libs/database/sources.cpp libs/database/database.cpp libs/database/insights.cpp libs/database/scaling.cpp libs/database/archive.cpp)
  target_link_libraries(t64storage PUBLIC t64core Qt6::Core Qt6::Sql PRIVATE SQLite::SQLite3)
  target_include_directories(t64storage PUBLIC libs/database)
  if(T64_BENCHMARKS)
    add_executable(database_benchmark tests/database_benchmark.cpp)
    target_link_libraries(database_benchmark PRIVATE t64storage)
    add_executable(pgn_benchmark tests/pgn_benchmark.cpp)
    target_link_libraries(pgn_benchmark PRIVATE t64storage)
    if(BUILD_TESTING)
      find_package(Python3 REQUIRED COMPONENTS Interpreter)
      add_test(NAME pgn_benchmark_safety COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_pgn_benchmark.py $<TARGET_FILE:pgn_benchmark>)
      set_tests_properties(pgn_benchmark_safety PROPERTIES TIMEOUT 60)
    endif()
  endif()
  add_library(t64engine libs/engine/uci.cpp libs/engine/comparison.cpp)
  target_link_libraries(t64engine PUBLIC t64core Qt6::Core)
  target_include_directories(t64engine PUBLIC libs/engine)
  add_library(t64analysis libs/analysis/queue.cpp)
  target_include_directories(t64analysis PUBLIC libs/analysis)
  target_link_libraries(t64analysis PUBLIC t64storage t64engine Qt6::Core)
  add_library(t64updates libs/updates/releases.cpp libs/updates/updater.cpp)
  target_include_directories(t64updates PUBLIC libs/updates)
  target_link_libraries(t64updates PUBLIC Qt6::Core Qt6::Network)
  add_library(t64content libs/content/content.cpp libs/content/package.cpp)
  target_include_directories(t64content PUBLIC libs/content)
  target_link_libraries(t64content PUBLIC Qt6::Core Qt6::Network Qt6::Concurrent PRIVATE LibArchive::LibArchive)
  qt_add_executable(trait64 app/main.cpp app/controller.cpp app/features.cpp app/appearance.cpp app/reliability.cpp app/content.cpp app/search.cpp app/imports.cpp app/repertoires.cpp app/repertoire_tasks.cpp app/preparation.cpp app/analyses.cpp)
  target_link_libraries(trait64 PRIVATE t64analysis t64storage t64engine t64updates t64content Qt6::Gui Qt6::Quick Qt6::QuickControls2 Qt6::Qml Qt6::Concurrent)
  file(GLOB_RECURSE T64_ASSETS CONFIGURE_DEPENDS assets/icons/*.svg assets/pieces/*.svg assets/openings/*.tsv assets/content/*)
  set(T64_UI translations/en.json ui/T64Sidebar.qml ui/T64Icon.qml ui/AnalysisQueue.qml ui/Preparation.qml ui/Repertoires.qml ui/Imports.qml ui/Downloads.qml ui/EngineOptions.qml ui/Home.qml ui/TTabButton.qml ui/Main.qml ui/Board.qml ui/TButton.qml ui/Piece.qml ui/Appearance.qml ui/PositionEditor.qml ui/Review.qml ui/Players.qml ui/Updates.qml ui/CompareEngines.qml)
  qt_add_resources(trait64 "app_resources" PREFIX "/" BASE "${CMAKE_CURRENT_SOURCE_DIR}" FILES ${T64_UI} ${T64_ASSETS} assets/trait64.svg assets/demo.pgn)
  if(BUILD_TESTING)
    add_executable(content_tests tests/content_tests.cpp)
    target_link_libraries(content_tests PRIVATE t64content t64engine Qt6::Test LibArchive::LibArchive)
    qt_add_resources(content_tests "content_test_resources" PREFIX "/" BASE "${CMAKE_CURRENT_SOURCE_DIR}" FILES assets/content/catalog.json assets/content/PlentyChess-LICENSE.txt)
    add_test(NAME content COMMAND content_tests)
    set_tests_properties(content PROPERTIES TIMEOUT 90)
    add_executable(update_tests tests/update_tests.cpp)
    target_link_libraries(update_tests PRIVATE t64updates Qt6::Test)
    add_test(NAME updates COMMAND update_tests)
    set_tests_properties(updates PROPERTIES TIMEOUT 30)
    add_executable(storage_tests tests/storage_tests.cpp)
    target_link_libraries(storage_tests PRIVATE t64storage Qt6::Test SQLite::SQLite3)
    add_test(NAME storage COMMAND storage_tests)
    set_tests_properties(storage PROPERTIES TIMEOUT 60)
    add_executable(repertoire_tests tests/repertoire_tests.cpp)
    target_link_libraries(repertoire_tests PRIVATE t64storage Qt6::Test)
    add_test(NAME repertoires COMMAND repertoire_tests)
    set_tests_properties(repertoires PROPERTIES TIMEOUT 60)
    add_executable(import_tests tests/import_tests.cpp)
    target_link_libraries(import_tests PRIVATE t64storage Qt6::Test)
    add_test(NAME resumable_import COMMAND import_tests)
    set_tests_properties(resumable_import PROPERTIES TIMEOUT 60)
    add_executable(analysis_tests tests/analysis_tests.cpp)
    target_link_libraries(analysis_tests PRIVATE t64analysis Qt6::Test)
    add_test(NAME persistent_analysis COMMAND analysis_tests)
    set_tests_properties(persistent_analysis PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 90)
    add_executable(engine_tests tests/engine_tests.cpp)
    target_link_libraries(engine_tests PRIVATE t64engine Qt6::Test)
    add_test(NAME engine COMMAND engine_tests)
    set_tests_properties(engine PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 60)
    add_test(NAME desktop_smoke COMMAND trait64 --smoke-test)
    set_tests_properties(desktop_smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software" TIMEOUT 20)
    qt_add_executable(desktop_tests tests/desktop_tests.cpp app/controller.cpp app/features.cpp app/appearance.cpp app/reliability.cpp app/content.cpp app/search.cpp app/imports.cpp app/repertoires.cpp app/repertoire_tasks.cpp app/preparation.cpp app/analyses.cpp)
    target_include_directories(desktop_tests PRIVATE app)
    target_link_libraries(desktop_tests PRIVATE t64analysis t64storage t64engine t64updates t64content Qt6::Gui Qt6::Quick Qt6::QuickControls2 Qt6::Qml Qt6::Concurrent Qt6::Test)
    qt_add_resources(desktop_tests "test_resources" PREFIX "/" BASE "${CMAKE_CURRENT_SOURCE_DIR}" FILES ${T64_UI} ${T64_ASSETS} assets/trait64.svg assets/demo.pgn)
    add_test(NAME desktop_layout COMMAND desktop_tests --ui-layout)
    set_tests_properties(desktop_layout PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software" TIMEOUT 120)
    add_test(NAME desktop_interactions COMMAND desktop_tests)
    set_tests_properties(desktop_interactions PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software" TIMEOUT 60)
  endif()
  install(TARGETS trait64 RUNTIME DESTINATION bin)
  install(FILES assets/trait64.desktop DESTINATION share/applications)
  install(FILES assets/trait64.svg DESTINATION share/icons/hicolor/scalable/apps)
  install(DIRECTORY assets/pieces/ DESTINATION share/doc/trait64/pieces FILES_MATCHING PATTERN "*.txt" PATTERN "*.md")
  install(FILES assets/openings/README.upstream.md assets/openings/LICENSE-CC0.txt DESTINATION share/doc/trait64/openings)
  install(FILES LICENSE THIRD_PARTY_NOTICES.md DESTINATION share/doc/trait64)
  set(CPACK_PACKAGE_NAME trait64)
  set(CPACK_PACKAGE_CONTACT "Sïmzy — Trait64")
  string(REPLACE "-" "~" CPACK_PACKAGE_VERSION "${T64_RELEASE_VERSION}")
  set(CPACK_GENERATOR "DEB;TGZ")
  set(CPACK_PACKAGING_INSTALL_PREFIX "/usr")
  set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
  set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
  set(CPACK_DEBIAN_PACKAGE_DEPENDS "qml6-module-qtquick, qml6-module-qtquick-controls, qml6-module-qtquick-layouts, qml6-module-qtquick-dialogs, qml6-module-qtquick-templates, qml6-module-qtquick-window, qml6-module-qtqml-workerscript, qml6-module-qtqml, qml6-module-qtqml-models, libqt6sql6-sqlite, fonts-dejavu-core, libqt6svg6")
  set(CPACK_DEBIAN_PACKAGE_RECOMMENDS "stockfish, qt6-wayland, policykit-1, gh")
  # Bookworm includes libqsvg in libqt6svg6; Trixie Qt 6.8 packages it separately.
  if(Qt6_VERSION VERSION_GREATER_EQUAL "6.8")
    string(APPEND CPACK_DEBIAN_PACKAGE_DEPENDS ", qt6-svg-plugins")
  endif()
  include(CPack)
endif()
