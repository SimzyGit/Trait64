cmake_minimum_required(VERSION 3.22)
project(Trait64 VERSION 0.3.0 LANGUAGES CXX)
set(T64_RELEASE_VERSION "${PROJECT_VERSION}-dev.1" CACHE STRING "Published semantic version")
add_compile_definitions(T64_VERSION="${T64_RELEASE_VERSION}")
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(T64_BUILD_GUI "Build the Qt desktop application" ON)
option(T64_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
option(T64_BENCHMARKS "Build opt-in database and real-PGN benchmarks" OFF)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-Wall -Wextra -Wpedantic -Wformat=2)
  # -Wshadow ne fait partie ni de -Wall ni de -Wextra, et c'est pourtant lui qui aurait
  # évité qu'une variable de boucle masque un drapeau dans la recherche d'ouvertures :
  # le test lisait alors un pointeur toujours non nul. MSVC le signalait déjà (C4456).
  # Réservé au code du logiciel : dans les tests, réutiliser un nom dans une portée
  # imbriquée est courant et sans danger, et seize avertissements noieraient le signal.
  set(T64_SHADOW_WARNING -Wshadow)
endif()
if(MSVC)
  # Sources are UTF-8 (accents, figurines); /Zc:__cplusplus reports the real standard.
  add_compile_options(/utf-8 /W4 /permissive- /Zc:__cplusplus /EHsc)
  add_compile_definitions(NOMINMAX _CRT_SECURE_NO_WARNINGS)
endif()
if(T64_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
find_package(Threads REQUIRED)
add_library(t64core libs/core/game.cpp libs/core/formats.cpp libs/core/parallel.cpp libs/core/explanation.cpp libs/core/tactics.cpp libs/core/polyglot.cpp libs/core/accuracy.cpp libs/core/hardware.cpp libs/core/moveorder.cpp)
target_link_libraries(t64core PUBLIC Threads::Threads)
target_compile_options(t64core PRIVATE ${T64_SHADOW_WARNING})
target_include_directories(t64core PUBLIC libs/core)
target_include_directories(t64core SYSTEM PUBLIC third_party/chess-library)
include(CTest)
if(BUILD_TESTING)
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE t64core)
  add_test(NAME core COMMAND core_tests)
  add_executable(robustness_tests tests/robustness_tests.cpp)
  target_link_libraries(robustness_tests PRIVATE t64core)
  add_test(NAME robustness COMMAND robustness_tests)
  if(T64_SANITIZERS)
    # The full mutation corpus is CPU-heavy in unoptimized ASan/UBSan builds.
    set_tests_properties(robustness PROPERTIES TIMEOUT 300)
  else()
    set_tests_properties(robustness PROPERTIES TIMEOUT 120)
  endif()
endif()
if(T64_BUILD_GUI)
  find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 Sql Concurrent Test Network)
  find_package(LibArchive REQUIRED)
  find_package(SQLite3 REQUIRED)
  set(CMAKE_AUTOMOC ON)
  add_library(t64storage libs/database/analyses.cpp libs/database/repertoires.cpp libs/database/imports.cpp libs/database/sources.cpp libs/database/database.cpp libs/database/insights.cpp libs/database/scaling.cpp libs/database/archive.cpp libs/database/positions.cpp libs/database/puzzles.cpp libs/database/croissant.cpp)
  target_link_libraries(t64storage PUBLIC t64core Qt6::Core Qt6::Sql PRIVATE SQLite::SQLite3)
  target_compile_options(t64storage PRIVATE ${T64_SHADOW_WARNING})
  target_include_directories(t64storage PUBLIC libs/database)
  if(T64_BENCHMARKS)
    add_executable(database_benchmark tests/database_benchmark.cpp)
    target_link_libraries(database_benchmark PRIVATE t64storage)
    add_executable(pgn_benchmark tests/pgn_benchmark.cpp)
    target_link_libraries(pgn_benchmark PRIVATE t64storage)
    if(BUILD_TESTING)
      find_package(Python3 REQUIRED COMPONENTS Interpreter)
      add_test(NAME pgn_benchmark_safety COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_pgn_benchmark.py $<TARGET_FILE:pgn_benchmark>)
      set_tests_properties(pgn_benchmark_safety PROPERTIES TIMEOUT 60)
      # Une seconde à peine, à la plus petite échelle acceptée : c'est l'absence de ce
      # test qui a laissé le banc de base planter pendant toute une migration de schéma.
      add_test(NAME database_benchmark_safety COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_database_benchmark.py $<TARGET_FILE:database_benchmark>)
      set_tests_properties(database_benchmark_safety PROPERTIES TIMEOUT 300)
      # Une phrase sans traduction retombe sur le français : l'interface anglaise se met
      # à parler français par endroits, et cela ne se voit qu'à l'usage.
      add_test(NAME translations COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_translations.py)
      set_tests_properties(translations PROPERTIES TIMEOUT 60)
      # Le --uninstall du script ne retirait que l'AppImage : qui avait posé le paquet
      # Debian s'entendait répondre « désinstallé » avec le logiciel toujours en place.
      add_test(NAME uninstall COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_uninstall.py)
      set_tests_properties(uninstall PROPERTIES TIMEOUT 120)
    endif()
  endif()
  add_library(t64engine libs/engine/uci.cpp libs/engine/comparison.cpp libs/engine/tournament.cpp
                        libs/engine/discovery.cpp)
  target_link_libraries(t64engine PUBLIC t64core Qt6::Core)
  target_compile_options(t64engine PRIVATE ${T64_SHADOW_WARNING})
  target_include_directories(t64engine PUBLIC libs/engine)
  add_library(t64analysis libs/analysis/queue.cpp)
  target_include_directories(t64analysis PUBLIC libs/analysis)
  target_link_libraries(t64analysis PUBLIC t64storage t64engine Qt6::Core)
  target_compile_options(t64analysis PRIVATE ${T64_SHADOW_WARNING})
  add_library(t64updates libs/updates/releases.cpp libs/updates/updater.cpp)
  target_include_directories(t64updates PUBLIC libs/updates)
  target_link_libraries(t64updates PUBLIC Qt6::Core Qt6::Network)
  target_compile_options(t64updates PRIVATE ${T64_SHADOW_WARNING})
  add_library(t64content libs/content/content.cpp libs/content/package.cpp libs/content/online.cpp libs/content/puzzles.cpp libs/content/fide.cpp libs/content/unpack.cpp libs/content/explorer.cpp)
  target_include_directories(t64content PUBLIC libs/content)
  target_link_libraries(t64content PUBLIC t64core Qt6::Core Qt6::Network Qt6::Concurrent PRIVATE LibArchive::LibArchive)
  target_compile_options(t64content PRIVATE ${T64_SHADOW_WARNING})
  qt_add_executable(trait64 app/main.cpp app/controller.cpp app/features.cpp app/appearance.cpp app/reliability.cpp app/content.cpp app/search.cpp app/imports.cpp app/repertoires.cpp app/repertoire_tasks.cpp app/preparation.cpp app/analyses.cpp app/puzzles.cpp app/accounts.cpp app/profiles.cpp)
  target_link_libraries(trait64 PRIVATE t64analysis t64storage t64engine t64updates t64content Qt6::Gui Qt6::Quick Qt6::QuickControls2 Qt6::Qml Qt6::Concurrent)
  target_compile_options(trait64 PRIVATE ${T64_SHADOW_WARNING})
  # Chaque système va chercher l'icône ailleurs : Linux dans le fichier .desktop, macOS
  # dans Resources à condition qu'Info.plist la nomme, Windows dans le binaire lui-même.
  if(APPLE)
    # Sans cela, macOS reçoit un exécutable nu : pas de nom lisible, pas d'icône, et
    # rien que macdeployqt puisse emporter. Le paquet est la forme attendue là-bas.
    target_sources(trait64 PRIVATE assets/trait64.icns)
    set_source_files_properties(assets/trait64.icns PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
    set_target_properties(trait64 PROPERTIES
      MACOSX_BUNDLE TRUE
      MACOSX_BUNDLE_BUNDLE_NAME "Trait64"
      MACOSX_BUNDLE_GUI_IDENTIFIER "org.trait64.Trait64"
      MACOSX_BUNDLE_ICON_FILE "trait64.icns"
      # Apple attend trois nombres dans la version courte ; le détail du canal de
      # développement tient dans l'autre, que le Finder affiche entre parenthèses.
      MACOSX_BUNDLE_BUNDLE_VERSION "${T64_RELEASE_VERSION}"
      MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}")
  elseif(WIN32)
    target_sources(trait64 PRIVATE assets/trait64.rc)
  endif()
  file(GLOB_RECURSE T64_ASSETS CONFIGURE_DEPENDS assets/icons/*.svg assets/app/*.png assets/flags/*.png assets/pieces/*.svg assets/openings/*.tsv assets/content/*)
  set(T64_UI ui/MoveNotebook.qml ui/TDialog.qml ui/TComboBox.qml translations/en.json ui/T64Sidebar.qml ui/T64Icon.qml ui/AnalysisQueue.qml ui/Preparation.qml ui/Repertoires.qml ui/Imports.qml ui/Downloads.qml ui/EngineOptions.qml ui/Home.qml ui/TTabButton.qml ui/Main.qml ui/Board.qml ui/TButton.qml ui/Piece.qml ui/Appearance.qml ui/PositionEditor.qml ui/Review.qml ui/Players.qml ui/Updates.qml ui/CompareEngines.qml ui/Tournament.qml ui/Puzzles.qml ui/Accounts.qml ui/ReviewImported.qml ui/AccuracyPanel.qml ui/Profiles.qml ui/SkillDashboard.qml ui/EvalBar.qml ui/EvalGraph.qml ui/SkillReport.qml ui/ProfilePage.qml ui/StudiesPage.qml ui/PlayerName.qml ui/Avatar.qml ui/MoveQuality.qml ui/OpeningReport.qml ui/GamePreview.qml ui/StudyExplorer.qml ui/OnlineExplorer.qml)
  qt_add_resources(trait64 "app_resources" PREFIX "/" BASE "${CMAKE_CURRENT_SOURCE_DIR}" FILES ${T64_UI} ${T64_ASSETS} assets/trait64.svg assets/demo.pgn)
  if(BUILD_TESTING)
    add_executable(content_tests tests/content_tests.cpp)
    target_link_libraries(content_tests PRIVATE t64content t64engine Qt6::Test LibArchive::LibArchive)

    set_tests_properties(online PROPERTIES TIMEOUT 60)
    set_tests_properties(content PROPERTIES TIMEOUT 90)

    add_executable(storage_tests tests/storage_tests.cpp)

    set_tests_properties(study_walk PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software" TIMEOUT 120)
    # Le budget d'une image vaut pour une machine au repos ; les exécuteurs partagés et
